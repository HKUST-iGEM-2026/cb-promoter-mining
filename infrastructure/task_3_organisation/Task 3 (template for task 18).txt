# ============================================================================
# Configuration
# ============================================================================
@dataclass
class ProkBertConfig:
    model_name: str = "biomedia-ai/prokbert-mini-long"
    max_length: int = 300
    batch_size: int = 16
    learning_rate: float = 2e-5
    num_epochs: int = 10
    warmup_ratio: float = 0.1
    weight_decay: float = 0.01
    dropout_rate: float = 0.1
    use_lora: bool = False  # ProkBERT is smaller, full fine-tuning is fine
    gradient_accumulation_steps: int = 4
    fp16: bool = True
    output_dir: str = "models/prokbert_promoter"
    seed: int = 42

class DNADataset(Dataset):
    """Dataset for DNA sequences with GC-content aware sampling"""
    
    def __init__(self, sequences, labels, tokenizer, max_length=300, 
                 augment=False, gc_bins=None):
        self.sequences = sequences
        self.labels = labels
        self.tokenizer = tokenizer
        self.max_length = max_length
        self.augment = augment
        self.gc_bins = gc_bins or [0.2, 0.3, 0.4, 0.5]
        
    def __len__(self):
        return len(self.sequences)
    
    def __getitem__(self, idx):
        seq = self.sequences[idx]
        label = self.labels[idx]
        
        # Data augmentation for promoter sequences
        if self.augment and label == 1 and np.random.random() > 0.7:
            seq = self._augment_sequence(seq)
        
        # Tokenize
        encoding = self.tokenizer(
            seq,
            truncation=True,
            padding='max_length',
            max_length=self.max_length,
            return_tensors='pt'
        )
        
        return {
            'input_ids': encoding['input_ids'].squeeze(),
            'attention_mask': encoding['attention_mask'].squeeze(),
            'labels': torch.tensor(label, dtype=torch.long)
        }
    
    def _augment_sequence(self, seq):
        """Reverse complement augmentation for promoters"""
        if np.random.random() > 0.5:
            # Reverse complement
            complement = {'A': 'T', 'T': 'A', 'G': 'C', 'C': 'G', 'N': 'N'}
            seq = ''.join(complement.get(base, base) for base in seq[::-1])
        return seq

class WeightedFocalLoss(nn.Module):
    """Focal Loss for imbalanced promoter/non-promoter classes"""
    
    def __init__(self, alpha=0.25, gamma=2.0, pos_weight=3.0):
        super().__init__()
        self.alpha = alpha
        self.gamma = gamma
        self.pos_weight = pos_weight
        
    def forward(self, inputs, targets):
        ce_loss = nn.functional.cross_entropy(
            inputs, targets, reduction='none'
        )
        pt = torch.exp(-ce_loss)
        focal_loss = self.alpha * (1-pt)**self.gamma * ce_loss
        
        # Apply positive class weighting
        weights = torch.ones_like(targets, dtype=torch.float)
        weights[targets == 1] = self.pos_weight
        focal_loss = focal_loss * weights
        
        return focal_loss.mean()

# ============================================================================
# Training Functions
# ============================================================================
def compute_metrics(eval_pred):
    """Compute metrics for evaluation"""
    predictions, labels = eval_pred
    predictions = np.argmax(predictions, axis=1)
    
    return {
        'accuracy': accuracy_score(labels, predictions),
        'f1': f1_score(labels, predictions, average='binary'),
        'auroc': roc_auc_score(labels, predictions)
    }

def objective(trial, train_dataset, val_dataset, config):
    """Optuna objective for hyperparameter optimization"""
    
    # Suggest hyperparameters
    learning_rate = trial.suggest_float('learning_rate', 1e-6, 5e-5, log=True)
    batch_size = trial.suggest_categorical('batch_size', [8, 16, 32])
    warmup_ratio = trial.suggest_float('warmup_ratio', 0.05, 0.15)
    weight_decay = trial.suggest_float('weight_decay', 0.0, 0.1)
    dropout_rate = trial.suggest_float('dropout_rate', 0.05, 0.3)
    
    # Update config
    config.learning_rate = learning_rate
    config.batch_size = batch_size
    config.warmup_ratio = warmup_ratio
    config.weight_decay = weight_decay
    config.dropout_rate = dropout_rate
    
    # Load model with custom dropout
    model = AutoModelForSequenceClassification.from_pretrained(
        config.model_name,
        num_labels=2,
        hidden_dropout_prob=dropout_rate,
        attention_probs_dropout_prob=dropout_rate
    )
    
    # Training arguments
    training_args = TrainingArguments(
        output_dir=f"{config.output_dir}/trial_{trial.number}",
        learning_rate=learning_rate,
        per_device_train_batch_size=batch_size,
        per_device_eval_batch_size=batch_size,
        num_train_epochs=config.num_epochs,
        weight_decay=weight_decay,
        warmup_ratio=warmup_ratio,
        logging_dir='./logs',
        logging_steps=50,
        evaluation_strategy="steps",
        eval_steps=200,
        save_strategy="steps",
        save_steps=500,
        load_best_model_at_end=True,
        metric_for_best_model="f1",
        greater_is_better=True,
        fp16=config.fp16,
        gradient_accumulation_steps=config.gradient_accumulation_steps,
        report_to="wandb",
        run_name=f"prokbert_trial_{trial.number}"
    )
    
    # Initialize trainer
    trainer = Trainer(
        model=model,
        args=training_args,
        train_dataset=train_dataset,
        eval_dataset=val_dataset,
        tokenizer=AutoTokenizer.from_pretrained(config.model_name),
        compute_metrics=compute_metrics,
        callbacks=[EarlyStoppingCallback(early_stopping_patience=3)]
    )
    
    # Train
    trainer.train()
    
    # Evaluate
    eval_results = trainer.evaluate()
    
    return eval_results['eval_f1']

def train_with_optuna(train_seqs, train_labels, val_seqs, val_labels, config, n_trials=20):
    """Run hyperparameter optimization with Optuna"""
    
    tokenizer = AutoTokenizer.from_pretrained(config.model_name)
    
    # Create datasets
    train_dataset = DNADataset(
        train_seqs, train_labels, tokenizer, 
        config.max_length, augment=True
    )
    val_dataset = DNADataset(
        val_seqs, val_labels, tokenizer, 
        config.max_length, augment=False
    )
    
    # Optuna study
    study = optuna.create_study(
        direction='maximize',
        study_name='prokbert_optimization',
        storage='sqlite:///prokbert_optuna.db',
        load_if_exists=True
    )
    
    study.optimize(
        lambda trial: objective(trial, train_dataset, val_dataset, config),
        n_trials=n_trials,
        n_jobs=1
    )
    
    # Train best model
    best_params = study.best_params
    logger.info(f"Best parameters: {best_params}")
    
    # Final training with best params
    final_model = train_final_model(
        train_seqs + val_seqs, 
        train_labels + val_labels,
        best_params,
        config
    )
    
    return final_model, study

def train_final_model(sequences, labels, best_params, config):
    """Train final model with best hyperparameters"""
    
    tokenizer = AutoTokenizer.from_pretrained(config.model_name)
    
    # Update config with best params
    for key, value in best_params.items():
        setattr(config, key, value)
    
    # Full dataset
    full_dataset = DNADataset(
        sequences, labels, tokenizer,
        config.max_length, augment=True
    )
    
    # Split for validation during final training
    train_size = int(0.9 * len(full_dataset))
    val_size = len(full_dataset) - train_size
    train_dataset, val_dataset = torch.utils.data.random_split(
        full_dataset, [train_size, val_size]
    )
    
    # Model with best dropout
    model = AutoModelForSequenceClassification.from_pretrained(
        config.model_name,
        num_labels=2,
        hidden_dropout_prob=config.dropout_rate,
        attention_probs_dropout_prob=config.dropout_rate
    )
    
    # Training arguments
    training_args = TrainingArguments(
        output_dir=config.output_dir,
        learning_rate=config.learning_rate,
        per_device_train_batch_size=config.batch_size,
        per_device_eval_batch_size=config.batch_size,
        num_train_epochs=config.num_epochs * 1.5,  # Train longer on full data
        weight_decay=config.weight_decay,
        warmup_ratio=config.warmup_ratio,
        logging_steps=50,
        evaluation_strategy="steps",
        eval_steps=200,
        save_strategy="epoch",
        load_best_model_at_end=True,
        metric_for_best_model="f1",
        fp16=config.fp16,
        gradient_accumulation_steps=config.gradient_accumulation_steps,
        save_total_limit=2,
        report_to="wandb",
        run_name="prokbert_final"
    )
    
    # Trainer with custom loss
    class CustomTrainer(Trainer):
        def compute_loss(self, model, inputs, return_outputs=False):
            labels = inputs.pop("labels")
            outputs = model(**inputs)
            logits = outputs.logits
            loss_fct = WeightedFocalLoss(pos_weight=3.0)
            loss = loss_fct(logits, labels)
            return (loss, outputs) if return_outputs else loss
    
    trainer = CustomTrainer(
        model=model,
        args=training_args,
        train_dataset=train_dataset,
        eval_dataset=val_dataset,
        tokenizer=tokenizer,
        compute_metrics=compute_metrics,
        callbacks=[EarlyStoppingCallback(early_stopping_patience=5)]
    )
    
    trainer.train()
    
    # Save model
    model.save_pretrained(f"{config.output_dir}/final_model")
    tokenizer.save_pretrained(f"{config.output_dir}/final_model")
    
    return model

# ============================================================================
# Main execution
# ============================================================================
def main():
    # Load data from previous tasks (T4, T16, T17)
    logger.info("Loading training data...")
    
    # Assuming data from previous tasks
    train_df = pd.read_parquet("data/train_promoters.parquet")
    val_df = pd.read_parquet("data/val_promoters.parquet")
    test_df = pd.read_parquet("data/test_promoters.parquet")
    
    # Extract sequences and labels
    train_seqs = train_df['sequence'].tolist()
    train_labels = train_df['is_promoter'].tolist()
    val_seqs = val_df['sequence'].tolist()
    val_labels = val_df['is_promoter'].tolist()
    
    # Initialize config
    config = ProkBertConfig()
    
    # Initialize wandb
    wandb.init(project="promoter_prediction", name="prokbert_training")
    
    # Run hyperparameter optimization
    logger.info("Starting hyperparameter optimization...")
    best_model, study = train_with_optuna(
        train_seqs, train_labels, 
        val_seqs, val_labels, 
        config, 
        n_trials=20
    )
    
    # Save study results
    study_results = pd.DataFrame(study.trials)
    study_results.to_csv("results/prokbert_optuna_results.csv", index=False)
    
    # Evaluate on test set
    logger.info("Evaluating on test set...")
    tokenizer = AutoTokenizer.from_pretrained(f"{config.output_dir}/final_model")
    model = AutoModelForSequenceClassification.from_pretrained(
        f"{config.output_dir}/final_model"
    )
    
    test_dataset = DNADataset(
        test_df['sequence'].tolist(),
        test_df['is_promoter'].tolist(),
        tokenizer,
        config.max_length,
        augment=False
    )
    
    trainer = Trainer(model=model, args=TrainingArguments(
        output_dir="./test_eval",
        per_device_eval_batch_size=config.batch_size,
    ))
    
    predictions = trainer.predict(test_dataset)
    test_metrics = compute_metrics((predictions.predictions, test_dataset.labels))
    
    logger.info(f"Test metrics: {test_metrics}")
    
    # Save metrics
    with open("results/prokbert_metrics.json", "w") as f:
        import json
        json.dump({
            "best_params": study.best_params,
            "test_metrics": test_metrics,
            "best_f1": study.best_value
        }, f, indent=2)
    
    logger.info("ProkBERT fine-tuning complete!")

if __name__ == "__main__":
    main()
